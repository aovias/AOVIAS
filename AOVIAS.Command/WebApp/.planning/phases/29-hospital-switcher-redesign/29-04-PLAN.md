import { Injectable, inject } from '@angular/core';
import { BehaviorSubject } from 'rxjs';
import { UserContextService } from './user-context.service';
import { ComplianceContextService } from './compliance-context.service';
import { AuthService } from '../auth/auth.service';
import { LoggerService } from './logger.service';

@Injectable({ providedIn: 'root' })
export class AppContextService {
  private userContext = inject(UserContextService);
  private complianceContext = inject(ComplianceContextService);
  private authService = inject(AuthService);
  private logger = inject(LoggerService);

  /** Emits the hospital ID every time a hospital switch completes */
  private hospitalSwitchedSubject = new BehaviorSubject<string | null>(null);
  readonly hospitalSwitched$ = this.hospitalSwitchedSubject.asObservable();

  /** Whether a hospital switch is currently in progress */
  private switchingSubject = new BehaviorSubject<boolean>(false);
  readonly switching$ = this.switchingSubject.asObservable();

  switchHospital(hospitalId: string): void {
    this.logger.info('Switching hospital context', hospitalId);
    this.switchingSubject.next(true);

    // 1. Update compliance context (triggers hospitalId$ subscribers)
    this.complianceContext.setHospitalId(hospitalId);

    // 2. Load framework for new hospital (future implementation)
    this.loadFrameworkForHospital(hospitalId);

    // 3. Signal switch complete
    this.hospitalSwitchedSubject.next(hospitalId);
    this.switchingSubject.next(false);

    this.logger.info('Hospital context switched', hospitalId);
  }

  loadFrameworkForHospital(hospitalId: string): void {
    // TODO: Load framework selection for hospital when framework API is available
    // For now, clear framework to trigger reload
    this.complianceContext.setFrameworkId(null);
  }

  initializeContext(): void {
    // Initialize context from stored data or API
  }

  initializeFromSession(): void {
    if (this.authService.isAuthenticated()) {
      this.initializeContext();
    }
  }
}

<task id="29-04-03" title="Update sidebar to react to hospital context changes">
**File:** `src/app/core/layout/sidebar.component.ts`

Ensure the sidebar hospital card stays in sync with the global hospital context:

1. **On initialization:**
   - Check `ComplianceContextService.getHospitalId()` for existing selection
   - If set, resolve hospital name from `OrganizationService.getHospitals()`
   - If not set, use `OrganizationService.getDefaultHospital()` and set it via `AppContextService.switchHospital()`

2. **Subscribe to `ComplianceContextService.hospitalId$`:**
   - On each emission, look up hospital by ID in `OrganizationService.getHospitals()`
   - Update `selectedHospitalName` and `selectedOrganizationName` signals
   - Use `takeUntilDestroyed()` for cleanup

3. **Handle edge cases:**
   - No hospitals available → show "No hospitals" / "Organization"
   - Hospital ID not found in list → show "Unknown hospital"
   - Organization not loaded yet → subscribe to `organizationService.loadOrganizationDetails()` first

```typescript
private readonly destroyRef = inject(DestroyRef);

ngOnInit(): void {
  this.initializeHospitalDisplay();
}

private initializeHospitalDisplay(): void {
  // React to hospital context changes
  this.complianceContext.hospitalId$
    .pipe(takeUntilDestroyed(this.destroyRef))
    .subscribe(hospitalId => {
      this.updateHospitalDisplay(hospitalId);
    });

  // Ensure organization is loaded
  const cachedOrg = this.organizationService.getOrganization();
  if (!cachedOrg) {
    this.organizationService.loadOrganizationDetails()
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe(() => {
        this.updateHospitalDisplay(this.complianceContext.getHospitalId());
      });
  }
}

private updateHospitalDisplay(hospitalId: string | null): void {
  const org = this.organizationService.getOrganization();
  const orgName = org?.organizationName ?? 'Organization';

  if (!hospitalId) {
    this.selectedHospitalName.set('No hospital selected');
    this.selectedOrganizationName.set(orgName);
    return;
  }

  const hospitals = this.organizationService.getHospitals();
  const hospital = hospitals.find(h => h.id === hospitalId);

  if (hospital) {
    this.selectedHospitalName.set(hospital.name);
    this.selectedOrganizationName.set(orgName);
  } else {
    this.selectedHospitalName.set('Unknown hospital');
    this.selectedOrganizationName.set(orgName);
  }
}
```
